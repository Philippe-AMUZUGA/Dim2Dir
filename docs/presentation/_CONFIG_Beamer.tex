% Division VERTICALE (fond uni, fraction 0-1 depuis le bas)
\newcommand{\specialpageV}[2]{%
    \begin{tikzpicture}[remember picture,overlay]
        \path (current page.north west) -- (current page.south west)
            coordinate[pos=#1] (divpoint);
        \fill[#2] (current page.south west) rectangle (current page.north east |- divpoint);
    \end{tikzpicture}%
}

% Division HORIZONTALE (fond uni, fraction 0-1 depuis la gauche)
\newcommand{\specialpageH}[2]{%
    \begin{tikzpicture}[remember picture,overlay]
        \path (current page.west) -- (current page.east)
            coordinate[pos=#1] (divpoint);
        \fill[#2] (current page.north west) rectangle (current page.south east -| divpoint);
    \end{tikzpicture}%
}

% Version opposée VERTICALE (colore le haut)
\newcommand{\specialpageVopp}[2]{%
    \begin{tikzpicture}[remember picture,overlay]
        \path (current page.south west) -- (current page.north west)
            coordinate[pos=#1] (divpoint);
        \fill[#2] (divpoint) rectangle (current page.north east);
    \end{tikzpicture}%
}

% Version opposée HORIZONTALE (colore la droite)
\newcommand{\specialpageHopp}[2]{%
    \begin{tikzpicture}[remember picture,overlay]
        \path (current page.east) -- (current page.west)
            coordinate[pos=#1] (divpoint);
        \fill[#2] (divpoint |- current page.north east) rectangle (current page.south east);
    \end{tikzpicture}%
}

% ================== Image + voile : variantes ==================

% Image clipée horizontalement (depuis la gauche) + voile
\newcommand{\specialimageH}[4]{%
    \begin{tikzpicture}[remember picture,overlay]
        \coordinate (clipEnd) at ($(current page.south west) + (#4*\paperwidth, 0)$);
        \begin{scope}
            \clip (current page.south west) rectangle (clipEnd |- current page.north east);
            \node[anchor=south west,inner sep=0pt] at (current page.south west) {%
                \includegraphics[width=\paperwidth,height=\paperheight,keepaspectratio]{#1}%
            };
        \end{scope}
        \begin{scope}
            \clip (current page.south west) rectangle (clipEnd |- current page.north east);
            \fill[#2,opacity=#3] (current page.south west) rectangle (current page.north east);
        \end{scope}
    \end{tikzpicture}%
}

% Image clipée VERTICALement (depuis le bas) + voile  <-- utilisée par le code source

                  ([yshift=#4\paperheight]current page.south west -| current page.north east);
            \node[anchor=south west,inner sep=0pt] at (current page.south west) {%
                \includegraphics[width=\paperwidth,height=\paperheight,keepaspectratio=false]{#1}%
            };
        \end{scope}
        \fill[#2,opacity=#3] (current page.south west) rectangle

    \end{tikzpicture}%
}

% Environnement checklist paramétrable
% Syntaxe : \begin{checklist}[<taille>][<marge>][<espacement>][<symbole_item>][<symbole_subitem>]
\NewDocumentEnvironment{checklist}{O{\scriptsize} O{1.5em} O{0.5ex} O{$\square$} O{$\square$}}{%
  \setbeamertemplate{itemize item}{#4}%      % #4 : Symbole pour les items principaux
  \setbeamertemplate{itemize subitem}{#5}%   % #5 : Symbole pour les sous-items
  \setlength{\leftmargini}{#2}%              % #2 : Marge gauche de la liste
  \setlength{\itemindent}{-0.5em}%           % Indentation négative pour alignement
  \setlength{\itemsep}{#3}%                  % #3 : Espace vertical entre les items
  \setlength{\parskip}{0pt}%                 % Pas d'espace supplémentaire entre paragraphes
  \begin{itemize}{#1}%                       % #1 : Taille de police des items
}{%
  \end{itemize}%
}
% Variables pour les symboles (optionnel - pour une cohérence globale)
\newcommand{\defaultcheckitem}{$\square$}
\newcommand{\defaultchecksubitem}{$\square$}
\newcommand{\checkeditem}{$\checkmark$}
\newcommand{\bulletitem}{$\bullet$}

% Environnement checklist_num paramétrable avec enumerate
% Définition corrigée de checklist_num
\NewDocumentEnvironment{checklist_num}{O{\scriptsize} O{1.5em} O{0.5ex} O{a} O{normal}}{%
  % #1 : taille police, #2 : marge, #3 : espacement, #4 : style enum, #5 : bold/normal
  \setlength{\leftmargini}{#2}%
  \setlength{\itemindent}{-0.5em}%
  \setlength{\itemsep}{#3}%
  \setlength{\parskip}{0pt}%
  
  % Styles disponibles dans Beamer :
  % a = lettres minuscules, A = lettres majuscules
  % i = chiffres romains minuscules, I = chiffres romains majuscules
  % 1 = chiffres arabes (défaut)
  
  \begin{enumerate}[#4]%
  \ifthenelse{\equal{#5}{bold}}{%
    \setbeamertemplate{enumerate items}{\textbf{\insertenumlabel.}}%
  }{%
    \setbeamertemplate{enumerate items}{\insertenumlabel.}%
  }%
}{%
  \end{enumerate}%
}

% --- 2) Code inline type Markdown : `Blabla`
% Usage : \mdcode{Blabla}
\newtcbox{\mdcode}{%
  on line,
  box align=base,
  nobeforeafter,
  colback=codebg,
  colframe=coderule,
  boxrule=0.35pt,
  arc=2pt,
  left=3pt,
  right=3pt,
  top=1pt,
  bottom=1pt,
  fontupper=\ttfamily\footnotesize
}

% --- 3) Bloc de code type Markdown : ``` ... ```
% Usage :
%   \begin{mdcodeblock}
%   Blabla
%   \end{mdcodeblock}
% Optionnel : \begin{mdcodeblock}[language=Python]
\lstdefinestyle{md_block}{
  inputencoding=utf8,
  upquote=true,
  showstringspaces=false,
  breaklines=true,
  columns=fullflexible,
  basicstyle=\ttfamily\scriptsize,
  numbers=none,
  frame=single,
  framerule=0.4pt,
  rulecolor=\color{coderule},
  backgroundcolor=\color{codebg},
  xleftmargin=0pt,
  framexleftmargin=0.4em
}

\lstnewenvironment{mdcodeblock}[1][]{%
  \lstset{style=md_block,#1}%
}{}
